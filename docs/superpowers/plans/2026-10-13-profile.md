# Profil neu: „Meine Events“ zuerst (D-08) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Die Profilseite zeigt links „Kommende Events“ (Event-Datum, Status, Slot, Check-in) und den „Verlauf“, rechts Profilkarte, Badges als Liste und die Einstellungen.

**Architecture:**
1. Server: `utils/profileEvents.js` (rein, ohne Datenbank) teilt die Anmeldungen eines Nutzers in `upcoming` und `history`, sortiert sie und kürzt jedes Event auf acht Felder. `GET /api/user/profile` liefert diese beiden Listen statt `registrations`.
2. Markup: `pages/profile-view.js` (rein, ohne DOM und ohne `fetch`) baut aus der Antwort HTML für Profilkarte, Zeilen, Verlauf und Badge-Liste. Datums-Kachel, Zeitzeile und Status-Chips kommen aus `events/cards.js`.
3. Seite: `pages/profile.js` lädt, füllt die Bereiche und hält, wie viele Verlaufs-Zeilen sichtbar sind. Der Abschnitt in `index.html` ist ein Raster mit vier Bereichen; der Block „PROFILE SECTION“ in `style.css` wird ersetzt.

**Tech Stack:** Node.js ≥ 22, Express 5, Vanilla-JS-SPA (native ES-Module, kein Build), Bootstrap 5.3.2 + Bootstrap Icons aus `public/vendor/`, `node:test` + `node:vm` (`--experimental-vm-modules`), Playwright (`npm run test:e2e`).

**Spec:** `docs/superpowers/specs/2026-10-13-profile-design.md`

**Branch:** `57-d-08-profil-neu-meine-events-zuerst` (Ticket #57), von `origin/main` nach dem Merge von D-16.

`test/profileEvents.test.js`:

```js
const { test } = require('node:test');
const assert = require('node:assert/strict');
const { buildProfileEvents, toProfileEntry } = require('../utils/profileEvents');

const NOW = new Date('2026-10-13T12:00:00.000Z');
const HOUR = 60 * 60 * 1000;
const at = (hours) => new Date(NOW.getTime() + hours * HOUR).toISOString();

const event = (title, fields = {}) => ({ title, status: 'active', startDate: at(48), endDate: at(50), ...fields });
const registration = (eventId, fields = {}) => ({ id: `reg-${eventId}`, eventId, userId: 'u1', status: 'accepted', createdAt: at(-72), ...fields });
const titles = (entries) => entries.map(entry => (entry.event ? entry.event.title : null));

/**
 * Both lists for events keyed by id. Every event gets one accepted
 * registration unless `registrations` lists its own.
 *
 * @param {Object<string, Object>} events -> Event data by id.
 * @param {Array<Object>} [registrations] -> Registrations in the order the database returns them.
 * @returns {{ upcoming: Array<Object>, history: Array<Object> }}
 */
function build(events, registrations = Object.keys(events).map(id => registration(id))) {
    return buildProfileEvents(registrations, new Map(Object.entries(events)), NOW);
}

test('an event that runs, lies ahead or has no date is upcoming, an ended one is history', () => {
    const { upcoming, history } = build({
        ahead: event('Ahead'),
        running: event('Running', { startDate: at(-1), endDate: at(1) }),
        openEnd: event('Open end', { startDate: at(-1), endDate: '' }),
        undated: event('Undated', { startDate: '', endDate: '' }),
        ended: event('Ended', { startDate: at(-5), endDate: at(-3) }),
        // Without an end an event lasts two hours
        longAgo: event('Long ago', { startDate: at(-3), endDate: '' })
    });
    assert.deepEqual(titles(upcoming), ['Running', 'Open end', 'Ahead', 'Undated']);
    assert.deepEqual(titles(history), ['Long ago', 'Ended']);
});

test('cancelled, archived and deleted events are history whatever their date says', () => {
    const { upcoming, history } = build(
        {
            cancelled: event('Cancelled', { status: 'cancelled' }),
            archived: event('Archived', { status: 'archived' }),
            draft: event('Draft', { status: 'draft' })
        },
        [registration('cancelled'), registration('archived'), registration('draft'), registration('gone')]
    );
    assert.deepEqual(titles(upcoming), ['Draft']);
    assert.deepEqual(titles(history), ['Cancelled', 'Archived', null]);
    assert.equal(history[2].id, 'reg-gone');
});

test('upcoming events come by start, undated ones last, equal starts by registration date', () => {
    const { upcoming } = build(
        {
            late: event('Late', { startDate: at(100), endDate: at(102) }),
            early: event('Early', { startDate: at(10), endDate: at(12) }),
            undated: event('Undated', { startDate: '', endDate: '' }),
            twinB: event('Twin B', { startDate: at(50), endDate: at(52) }),
            twinA: event('Twin A', { startDate: at(50), endDate: at(52) }),
            undated2: event('Undated 2', { startDate: '', endDate: '' })
        },
        [
            registration('late'),
            registration('undated', { createdAt: at(-10) }),
            registration('twinB', { createdAt: at(-5) }),
            registration('early'),
            registration('twinA', { createdAt: at(-20) }),
            registration('undated2', { createdAt: at(-30) })
        ]
    );
    assert.deepEqual(titles(upcoming), ['Early', 'Twin A', 'Twin B', 'Late', 'Undated 2', 'Undated']);
});

test('the history starts with the latest event, a deleted event counts by its registration date', () => {
    const { history } = build(
        {
            old: event('Old', { startDate: at(-500), endDate: at(-498) }),
            recent: event('Recent', { startDate: at(-30), endDate: at(-28) })
        },
        [registration('old'), registration('gone', { createdAt: at(-100) }), registration('recent')]
    );
    assert.deepEqual(history.map(entry => entry.id), ['reg-recent', 'reg-gone', 'reg-old']);
});

test('every registration status is listed, rejected ones too', () => {
    const statuses = ['accepted', 'pending', 'waitlisted', 'rejected'];
    const events = Object.fromEntries(statuses.map((status, index) => [status, event(status, { startDate: at(10 + index), endDate: at(20) })]));
    const { upcoming, history } = build(events, statuses.map(status => registration(status, { status })));
    assert.deepEqual(upcoming.map(entry => entry.status), statuses);
    assert.deepEqual(history, []);
});

test('the slot is the one taken, the wish of an open registration, else null', () => {
    const raid = event('Raid', { slots: [{ id: 'tank', name: 'Tank', emoji: '🛡️', capacity: 1 }, { id: 'dps', name: 'Schaden', capacity: 3 }] });
    const slotOf = (fields) => toProfileEntry(registration('raid', fields), raid).slot;

    assert.deepEqual(slotOf({ slotId: 'tank', preferredSlotId: 'dps' }), { name: 'Tank', emoji: '🛡️', wish: false });
    assert.deepEqual(slotOf({ slotId: 'dps' }), { name: 'Schaden', emoji: '', wish: false });
    assert.deepEqual(slotOf({ status: 'waitlisted', slotId: null, preferredSlotId: 'tank' }), { name: 'Tank', emoji: '🛡️', wish: true });
    assert.deepEqual(slotOf({ status: 'pending', preferredSlotId: 'dps' }), { name: 'Schaden', emoji: '', wish: true });
    // Rejected: the wish no longer matters. A slot that was removed from the event is gone.
    assert.equal(slotOf({ status: 'rejected', slotId: null, preferredSlotId: 'tank' }), null);
    assert.equal(slotOf({ slotId: 'healer' }), null);
    assert.equal(slotOf({}), null);
    // An event without slots, or with broken slot data, has none
    assert.equal(toProfileEntry(registration('plain', { slotId: 'tank' }), event('Plain')).slot, null);
    assert.equal(toProfileEntry(registration('odd', { slotId: 'tank' }), event('Odd', { slots: 'tank' })).slot, null);
});

test('an entry carries eight event fields and never the check-in code', () => {
    const full = event('Full', {
        description: 'Geheim', formFields: [{ label: 'Name' }], customBotMessages: { accepted: 'Hi' }, category: 'Gaming',
        location: { text: 'Lobby', channelId: '1', channelType: 'voice' }, checkinEnabled: true, checkinCode: '123456'
    });
    const entry = toProfileEntry(registration('full', { checkedIn: true, formAnswers: { Name: 'Tom' }, guests: ['u2'], userName: 'Tom' }), full);
    assert.deepEqual(entry, {
        id: 'reg-full', status: 'accepted', checkedIn: true, createdAt: at(-72), slot: null,
```

```js
/**
 * The two lists of the profile page: what a member is registered for next
 * and what lies behind them. Pure helpers without database access;
 * routes/user.js loads registrations and events. "Over" follows the rule of
 * the event overview (isEventOver in eventListing.js).
 */
const { isEventOver, parseDate } = require('./eventListing');
const { getSlots, findSlot } = require('./slotService');

// Event statuses that end the waiting, whatever the date says
const CLOSED_STATUSES = ['cancelled', 'archived'];
// Registrations without a slot that still name the one they hope for
const WISH_STATUSES = ['pending', 'waitlisted'];

/**
 * Slot shown next to a registration.
 *
 * @param {Object} registration -> Registration data.
 * @param {Object} event -> Event data.
 * @returns {{ name: string, emoji: string, wish: boolean }|null} -> The slot taken; for a pending or waitlisted registration without one its main role with `wish: true`; null without slots or when the slot no longer exists.
 */
function profileSlot(registration, event) {
    const slots = getSlots(event);
    const describe = (slot, wish) => ({ name: String(slot.name || ''), emoji: String(slot.emoji || ''), wish });
    const taken = registration.slotId ? findSlot(slots, registration.slotId) : null;
    if (taken) return describe(taken, false);
    if (!WISH_STATUSES.includes(registration.status) || !registration.preferredSlotId) return null;
    const wished = findSlot(slots, registration.preferredSlotId);
    return wished ? describe(wished, true) : null;
}

/**
 * One entry of the profile lists: the registration and the few event fields the page shows.
 *
 * @param {Object} registration -> Registration with `id` and `eventId`.
 * @param {Object|null} event -> Event data, null when the event was deleted.
 * @returns {Object} -> `{ id, status, checkedIn, createdAt, slot, event }`; `event` is `{ id, title, startDate, endDate, status, category, location, checkinEnabled }` or null. Never carries the check-in code.
 */
function toProfileEntry(registration, event) {
    return {
        id: registration.id,
        status: registration.status || '',
        checkedIn: registration.checkedIn === true,
        createdAt: registration.createdAt || '',
        slot: event ? profileSlot(registration, event) : null,
        event: event ? {
            id: registration.eventId,
            title: event.title || '',
            startDate: event.startDate || '',
            endDate: event.endDate || '',
            status: event.status || '',
            category: event.category || '',
            location: String((event.location && event.location.text) || ''),
            // A check-in needs the switch and a generated code (POST /api/events/:id/checkin)
            checkinEnabled: Boolean(event.checkinEnabled && event.checkinCode)
        } : null
    };
}

/**
 * Start of an entry's event.
 *
 * @param {Object} entry -> Profile entry.
 * @returns {number|null} -> Milliseconds, or null without an event or a readable start.
 */
function startTime(entry) {
    const start = entry.event ? parseDate(entry.event.startDate) : null;
    return start ? start.getTime() : null;
}

/**
 * Day an entry was registered.
 *
 * @param {Object} entry -> Profile entry.
 * @returns {number} -> Milliseconds, 0 without a readable date.
 */
function createdTime(entry) {
    const created = parseDate(entry.createdAt);
    return created ? created.getTime() : 0;
}

// Next event first; events without a date go to the end; equal starts keep the order of registering
function byStartAscending(a, b) {
    const startA = startTime(a);
    const startB = startTime(b);
    if (startA === null || startB === null) {
        if (startA !== startB) return startA === null ? 1 : -1;
        return createdTime(a) - createdTime(b);
    }
    return startA - startB || createdTime(a) - createdTime(b);
}

// Latest event first; without an event or a start the registration date stands in
function byDateDescending(a, b) {
    return (startTime(b) ?? createdTime(b)) - (startTime(a) ?? createdTime(a));
}

/**
 * Split a member's registrations into what is coming up and what is over.
 *
 * @param {Array<Object>} registrations -> Registrations with `id`, every status included.
 * @param {Map<string, Object>} eventsById -> Event data by id; deleted events are missing.
 * @param {Date} [now] -> Reference time.
 * @returns {{ upcoming: Array<Object>, history: Array<Object> }} -> Entries of toProfileEntry(). `history` holds deleted, cancelled, archived and ended events, latest first; `upcoming` everything else by start.
 */
function buildProfileEvents(registrations, eventsById, now = new Date()) {
    const upcoming = [];
    const history = [];
    for (const registration of registrations) {
        const event = eventsById.get(registration.eventId) || null;
        const closed = !event || CLOSED_STATUSES.includes(event.status) || isEventOver(event, now);
        (closed ? history : upcoming).push(toProfileEntry(registration, event));
    }
    return { upcoming: upcoming.sort(byStartAscending), history: history.sort(byDateDescending) };
}

module.exports = { buildProfileEvents, toProfileEntry };
```

```js
/**
 * Time line of a card.
 *
 * @param {Object} event -> Event.
 * @param {Date} now -> Reference time.
 * @param {string} [place] -> Place of the event; appended when given (profile rows).
 * @returns {string} -> "Läuft gerade · seit 14:30", "Morgen · 19:00 · Lobby" … or "Termin folgt".
 */
export function renderEventWhen(event, now, place = '') {
    const start = parseDate(event.startDate);
    const where = place ? ` · ${escapeHtml(String(place))}` : '';
    if (!start) return `<p class="event-card-when">Termin folgt${where}</p>`;
    if (isRunning(event, now)) return `<p class="event-card-when is-live"><strong>Läuft gerade</strong> · seit ${formatTime(start)}${where}</p>`;
    return `<p class="event-card-when"><strong>${relativeDay(start, now)}</strong> · ${formatTime(start)}${where}</p>`;
}
```

```js
/**
 * Markup of the profile page (#profile): pure functions that turn the answer
 * of GET /api/user/profile into HTML. No DOM access and no fetch, so the
 * tests call them directly. Styles: PROFILE SECTION in style.css; date tile,
 * chips, avatar, skeleton and the progress bar come from components.css.
 */
import { avatar } from '../core/components.js';
import { escapeHtml } from '../core/ui.js';
import { eventStart, renderDateTile, renderEventWhen, renderStatusChip } from '../events/cards.js';

const HEX_COLOR = /^#[0-9a-fA-F]{6}$/;
const ICON_CLASS = /^bi-[a-z0-9-]+$/;
// Chip of a history row per registration status; cancelled events and check-ins come first (historyChip)
const HISTORY_LABELS = { accepted: 'Teilgenommen', rejected: 'Abgelehnt', waitlisted: 'Warteliste', pending: 'Nicht bestätigt' };
const CHECKED_IN = { tone: 'success', icon: 'qr-code-scan' };

/**
 * A small chip.
 *
 * @param {string} label -> Text, already escaped by the caller.
 * @param {Object} [options] -> `tone`: success, warning, danger, info or accent; `icon`: Bootstrap icon name without "bi-".
 * @returns {string} -> HTML of one chip.
 */
function chip(label, { tone = '', icon = '' } = {}) {
    return `<span class="chip chip-sm${tone ? ` chip-${tone}` : ''}">${icon ? `<i class="bi bi-${icon}"></i>` : ''}${label}</span>`;
}

/**
 * Chip for the slot of a registration.
 *
 * @param {Object|null} slot -> `{ name, emoji, wish }` of a profile entry.
 * @returns {string} -> "🛡️ Tank", for a wish "Wunsch: 🛡️ Tank"; empty without a named slot.
 */
function slotChip(slot) {
    if (!slot || !slot.name) return '';
    const label = `${slot.emoji ? `${escapeHtml(String(slot.emoji))} ` : ''}${escapeHtml(String(slot.name))}`;
    return chip(slot.wish ? `Wunsch: ${label}` : label);
}

/**
 * Profile card: avatar, name, handle and three numbers.
 *
 * @param {Object} user -> Session user (`displayName`, `username`, `avatar`).
 * @param {Object|null} stats -> `stats` of the profile answer; null while it loads or after an error shows dashes.
 * @returns {string} -> HTML for #profile-card.
 */
export function renderProfileCard(user, stats) {
    const numbers = [['Anmeldungen', 'total'], ['Teilnahmen', 'accepted'], ['Check-ins', 'checkedIn']];
    const cells = numbers.map(([label, key]) => {
        const value = stats ? Math.max(0, Math.floor(Number(stats[key]) || 0)) : '–';
        return `<div class="profile-number"><dt>${label}</dt><dd>${value}</dd></div>`;
    }).join('');
    return '<div class="profile-card-head">'
        + avatar({ url: user?.avatar, name: user?.displayName, size: 'lg' })
        + `<div class="profile-card-names"><div class="profile-name">${escapeHtml(String(user?.displayName ?? ''))}</div>`
        + `<div class="profile-handle">@${escapeHtml(String(user?.username ?? ''))}</div></div></div>`
        + `<dl class="profile-numbers">${cells}</dl>`;
}

/**
 * Row of "Kommende Events": date tile, title, time line with the place, chips,
 * and on the right an arrow or the check-in button.
 *
 * @param {Object} entry -> Entry of `upcoming` (with `event`).
 * @param {Date} now -> Reference time for "Heute" and "Läuft gerade".
 * @param {boolean} canOpen -> Whether the user may open the event page; false renders no link and nothing to click.
 * @returns {string} -> HTML of one .profile-event.
 */
export function renderUpcomingRow(entry, now, canOpen) {
    const event = entry.event || {};
    const rejected = entry.status === 'rejected';
    const checkedIn = entry.status === 'accepted' && entry.checkedIn === true;
    const checkinOpen = entry.status === 'accepted' && !checkedIn && event.checkinEnabled === true;

    let status = renderStatusChip(entry.status);
    if (checkedIn) status = chip('Eingecheckt', CHECKED_IN);
    if (rejected) status = chip('Abgelehnt', { tone: 'danger', icon: 'x-circle' });

    let action = '';
    if (canOpen && checkinOpen) action = '<span class="btn btn-accent btn-sm profile-event-action"><i class="bi bi-qr-code-scan me-1"></i>Einchecken</span>';
    else if (canOpen) action = '<i class="bi bi-chevron-right profile-event-action" aria-hidden="true"></i>';

    const inner = renderDateTile(event, now)
        + `<div class="profile-event-body"><h4 class="profile-event-title">${escapeHtml(String(event.title ?? ''))}</h4>`
        + renderEventWhen(event, now, event.location)
        + `<div class="profile-event-facts">${status}${rejected ? '' : slotChip(entry.slot)}</div></div>`
        + action;
    const classes = `glass-card profile-event${rejected ? ' is-muted' : ''}`;
    return canOpen
        ? `<a class="${classes}" href="#event/${escapeHtml(String(event.id ?? ''))}">${inner}</a>`
        : `<div class="${classes}">${inner}</div>`;
}

/**
 * Chip of a history row; the first matching case wins.
 *
 * @param {Object} entry -> Entry of `history`.
 * @returns {string} -> "Abgesagt", "Eingecheckt", or the label of the registration status; empty for an unknown status.
 */
function historyChip(entry) {
    if (entry.event && entry.event.status === 'cancelled') return chip('Abgesagt');
    if (entry.status === 'accepted' && entry.checkedIn === true) return chip('Eingecheckt', CHECKED_IN);
    return Object.hasOwn(HISTORY_LABELS, entry.status) ? chip(HISTORY_LABELS[entry.status]) : '';
}

/**
 * Day shown in a history row.
 *
 * @param {Object} entry -> Entry of `history`.
 * @returns {string} -> "06. Okt. 2026" for the start of the event, for a deleted event the day of the registration; "–" without a readable date.
 */
function historyDay(entry) {
    const date = eventStart({ startDate: entry.event ? entry.event.startDate : entry.createdAt });
    return date ? date.toLocaleDateString('de-DE', { day: '2-digit', month: 'short', year: 'numeric' }) : '–';
}

/**
 * Row of "Verlauf": day, title and one chip.
 *
 * @param {Object} entry -> Entry of `history`; `event` is null for a deleted event.
 * @param {boolean} canOpen -> Whether the user may open the event page.
 * @returns {string} -> HTML of one .profile-history-row, a link when the event exists and can be opened.
 */
export function renderHistoryRow(entry, canOpen) {
    const title = entry.event
        ? escapeHtml(String(entry.event.title ?? ''))
        : '<span class="text-muted">Gelöschtes Event</span>';
    const inner = `<span class="profile-history-date">${historyDay(entry)}</span>`
        + `<span class="profile-history-title">${title}</span>${historyChip(entry)}`;
    return canOpen && entry.event
        ? `<a class="profile-history-row" href="#event/${escapeHtml(String(entry.event.id ?? ''))}">${inner}</a>`
        : `<div class="profile-history-row">${inner}</div>`;
}

/**
 * Badge list: reached badges first, each with icon, name and rule; open ones
 * with a bar when they carry a progress value.
 *
 * @param {Array<Object>} badges -> `badges` of the profile answer.
 * @returns {string} -> `<li>` rows for #profile-badges; call applyStyleData() afterwards, it sets the colour of reached badges.
 */
export function renderBadgeList(badges) {
    const list = Array.isArray(badges) ? badges : [];
    const ordered = [...list.filter(badge => badge.unlocked), ...list.filter(badge => !badge.unlocked)];
    return ordered.map(badge => {
        const icon = ICON_CLASS.test(badge.icon) ? badge.icon : 'bi-award';
        const colour = badge.unlocked && HEX_COLOR.test(badge.color) ? ` data-style-color="${badge.color}"` : '';
        const rule = escapeHtml(String(badge.description ?? ''));
        const progress = !badge.unlocked && typeof badge.progress === 'number'
            ? `<progress class="capacity-meter" max="100" value="${Math.round(Math.min(1, Math.max(0, badge.progress)) * 100)}" aria-label="Fortschritt: ${rule}"></progress>`
            : '';
        return `<li class="badge-row${badge.unlocked ? '' : ' is-locked'}">`
            + `<span class="badge-row-icon"${colour}><i class="bi ${icon}"></i></span>`
            + `<div class="badge-row-body"><div class="badge-row-name">${escapeHtml(String(badge.name ?? ''))}</div>`
            + `<div class="badge-row-rule">${rule}</div>${progress}</div></li>`;
    }).join('');
}

/**
 * Placeholder rows of "Kommende Events" while the profile loads.
 *
 * @param {number} count -> Number of rows.
 * @returns {string} -> HTML of `count` skeleton rows in the row layout.
 */
export function renderProfileSkeleton(count) {
    const row = '<div class="glass-card profile-event" aria-hidden="true">'
        + '<div class="skeleton event-card-skeleton-date"></div>'
        + '<div class="profile-event-body"><div class="skeleton skeleton-title"></div><div class="skeleton skeleton-text is-short"></div></div>'
        + '</div>';
    return row.repeat(count);
}

/**
 * Placeholder rows of the badge list while the profile loads.
 *
 * @param {number} count -> Number of rows.
 * @returns {string} -> HTML of `count` skeleton rows.
 */
export function renderBadgeSkeleton(count) {
    const row = '<li class="badge-row" aria-hidden="true"><span class="skeleton skeleton-avatar"></span>'
        + '<div class="badge-row-body"><div class="skeleton skeleton-text is-short"></div></div></li>';
    return row.repeat(count);
}
```

```js
 * Fill card, upcoming events, badges and history from a profile answer.
 * Missing lists count as empty, so an incomplete answer still shows a page.
 *
 * @param {Object} data -> Answer of GET /api/user/profile.
 * @returns {void}
 */
function renderProfile(data) {
    const now = new Date();
    const upcoming = Array.isArray(data.upcoming) ? data.upcoming : [];
    const badges = Array.isArray(data.badges) ? data.badges : [];
    profileHistory = Array.isArray(data.history) ? data.history : [];
    historyVisible = HISTORY_PAGE_SIZE;

    document.getElementById('profile-card').innerHTML = renderProfileCard(state.currentUser, data.stats || {});
    document.getElementById('profile-upcoming-count').textContent = upcoming.length ? String(upcoming.length) : '';
    document.getElementById('profile-upcoming').innerHTML = upcoming.length
        ? upcoming.map(entry => renderUpcomingRow(entry, now, canOpenEvent(entry.event))).join('')
        : emptyState({
            icon: 'calendar-x',
            title: 'Noch nichts geplant',
            text: 'Melde dich für ein Event an, dann steht es hier.',
            action: '<a class="btn btn-accent" href="#events"><i class="bi bi-calendar-event me-1"></i>Events ansehen</a>'
        });

    const badgesEl = document.getElementById('profile-badges');
    badgesEl.innerHTML = renderBadgeList(badges);
    applyStyleData(badgesEl);
    document.getElementById('profile-badge-count').textContent = `${badges.filter(badge => badge.unlocked).length} von ${badges.length}`;

    renderProfileHistory();
}

/**
 * Show the first `historyVisible` rows of the history; without a history the block stays hidden.
 *
 * @returns {void}
 */
function renderProfileHistory() {
    toggleElement(document.getElementById('profile-history-block'), profileHistory.length > 0);
    document.getElementById('profile-history-count').textContent = profileHistory.length ? String(profileHistory.length) : '';
    document.getElementById('profile-history').innerHTML = profileHistory.slice(0, historyVisible)
        .map(entry => renderHistoryRow(entry, canOpenEvent(entry.event))).join('');
    toggleElement(document.getElementById('profile-history-more'), historyVisible < profileHistory.length);
}

/**
 * Show five more rows of the history; the rows are already loaded.
 *
 * @returns {void}
 */
export function showMoreProfileHistory() {
    historyVisible = Math.min(historyVisible + HISTORY_PAGE_SIZE, profileHistory.length);
    renderProfileHistory();
}
```
